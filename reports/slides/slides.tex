% ── MOTIVATION ──
\section{Motivation}
\begin{frame}{The Surveillance Problem}
\begin{columns}[T]
\column{0.52\textwidth}
\vspace{0.5em}
\textbf{900 million} surveillance cameras worldwide.\\
\vspace{0.4em}
A human operator watching \textbf{multiple feeds} experiences \alert{vigilance decrement} after approximately 20 minutes.\\
\vspace{0.4em}
Critical events go unnoticed. A missed event could mean a missed crime.

\column{0.48\textwidth}
\begin{block}{Three challenges}
\begin{enumerate}
  \item Visual events need \textbf{cross-frame tracking} to form temporal intervals
  \item Audio captures \textbf{off-camera} events invisible to cameras
  \item Combining modalities naively (temporal joins) \textbf{reduces recall}
\end{enumerate}
\end{block}
\end{columns}
\end{frame}

% ── PROBLEM & GAP ──
\section{Problem and Gap}
\begin{frame}{Prior Work: VIS MODE and Its Limitations}
\begin{block}{VIS MODE (Crescitelli et al., CIKM '26)}
First ISEQL-based event detector using VLM perception. Reported high precision/recall on surveillance footage.
\end{block}
\vspace{0.4em}
\begin{alertblock}{Three problems identified}
\begin{enumerate}
  \item \textbf{Zero-event bug:} VLM assigned new IDs every frame, intervals never spanned multiple frames, \alert{zero events detected}.
  \item \textbf{Unverifiable claims:} Evaluated on unlabeled ground truth.
  \item \textbf{Non-deterministic:} Temperature $>0$, no fixed seed.
\end{enumerate}
\end{alertblock}
\vspace{0.3em}
\textbf{Our fix:} Object re-identification via two-phase VLM tracking prompt.\hspace{1em}\textbf{Our extension:} Audio modality via LALM.
\end{frame}

% ── ISEQL FORMAL EXPRESSIONS (1/2) ──
\section{ISEQL}
\begin{frame}{ISEQL: Formal Event Definitions (1/2)}
\footnotesize
\textbf{Notation:} $M_i.\texttt{arg}_j$: $j$-th argument of interval $i$. $M_i.s_f$, $M_i.e_f$: start/end frame. $\Join_{\mathrm{Bef}_{\delta}}$: before operator with gap $\le\delta$. $\cup$: set union. $\sigma_P$: select intervals satisfying $P$. $\pi_F$: project fields $F$.
\vspace{0.3em}

\noindent\textbf{Fight} \textit{(visual + audio)}
\[
\begin{aligned}
\text{Fight}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.\texttt{arg}_2, M_1.s_f, M_1.e_f}(
  \sigma_{\texttt{pred}=\text{`physical\_altercation'}}(M_1) ) \\[2pt]
\text{Fight}_B &\equiv \pi_{M_1.s_f, M_1.e_f, M_2.s_f, M_2.e_f}(
  \sigma_{\texttt{pred}=\text{`shout'}}(M_1) \Join_{\mathrm{Bef}_{120}} \sigma_{\texttt{pred}=\text{`impact'}}(M_2) ) \\[2pt]
\text{Fight}_C &\equiv \text{Fight}_A \cup \text{Fight}_B
\end{aligned}
\]

\noindent\textbf{Handoff} \textit{(visual only, $\delta=240$)}
\[
\begin{aligned}
\text{Handoff}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.\texttt{arg}_2, M_1.s_f, M_1.e_f, M_2.\texttt{arg}_1, M_2.\texttt{arg}_2, M_2.s_f, M_2.e_f}( \\
&\qquad \sigma_{M_1.\texttt{arg}_1 \neq M_2.\texttt{arg}_1 \land M_1.\texttt{arg}_2 = M_2.\texttt{arg}_2}(
  \sigma_{\texttt{pred}=\text{`carrying'}}(M_1) \Join_{\mathrm{Bef}_{240}} \sigma_{\texttt{pred}=\text{`carrying'}}(M_2) ) )
\end{aligned}
\]

\noindent\textbf{Loitering} \textit{(visual only, duration $\ge 150$ frames)}
\[
\begin{aligned}
\text{Loitering}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.\texttt{arg}_2, M_1.s_f, M_1.e_f}(
  \sigma_{\texttt{pred}=\text{`suspicious\_near\_vehicle'}}(M_1) \land (M_1.e_f - M_1.s_f) \ge 150 )
\end{aligned}
\]
\end{frame}

% ── ISEQL FORMAL EXPRESSIONS (2/2) ──
\begin{frame}{ISEQL: Formal Event Definitions (2/2)}
\footnotesize
\noindent\textbf{Vehicle Escape} \textit{(visual + audio)}
\[
\begin{aligned}
\text{Escape}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.s_f, M_1.e_f, M_2.\texttt{arg}_1, M_2.\texttt{arg}_2, M_2.s_f, M_2.e_f}( \\
&\qquad \sigma_{M_1.\texttt{arg}_1 = M_2.\texttt{arg}_1}(
  \sigma_{\texttt{pred}=\text{`running'}}(M_1) \Join_{\mathrm{Bef}_{50}\lor\mathrm{Overlap}} \sigma_{\texttt{pred}=\text{`enter\_or\_exit\_vehicle'}}(M_2) ) ) \\[2pt]
\text{Escape}_B &\equiv \pi_{M_1.s_f, M_1.e_f, M_2.s_f, M_2.e_f}(
  \sigma_{\texttt{pred}\in\{\text{`engine'},\text{`vehicle'}\}}(M_1) \Join_{\mathrm{Bef}_{150}\lor\mathrm{Overlap}} \sigma_{\texttt{pred}=\text{`tire\_squeal'}}(M_2) ) \\[2pt]
\text{Escape}_C &\equiv \text{Escape}_A \cup \text{Escape}_B
\end{aligned}
\]

\noindent\textbf{Vehicle Collision} \textit{(visual + audio, $\delta=60$)}
\[
\begin{aligned}
\text{Collision}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.s_f, M_1.e_f}(
  \sigma_{\texttt{pred}=\text{`vehicle\_collision'}}(M_1) ) \\[2pt]
\text{Collision}_B &\equiv \pi_{M_1.s_f, M_1.e_f, M_2.s_f, M_2.e_f}(
  \sigma_{\texttt{pred}\in\{\text{`horn'},\text{`skidding'}\}}(M_1) \Join_{\mathrm{Bef}_{60}} \sigma_{\texttt{pred}\in\{\text{`impact'},\text{`glass\_breaking'}\}}(M_2) ) \\[2pt]
\text{Collision}_C &\equiv \text{Collision}_A \cup \text{Collision}_B
\end{aligned}
\]

\noindent\textbf{Gunshot / Explosion} \textit{(visual + audio)}
\[
\begin{aligned}
\text{Gunshot}_A &\equiv \pi_{M_1.\texttt{arg}_1, M_1.s_f, M_1.e_f}(
  \sigma_{\texttt{pred}\in\{\text{`gunshot\_visible'},\text{`explosion\_visible'}\}}(M_1) ) \\[2pt]
\text{Gunshot}_B &\equiv \pi_{M_1.s_f, M_1.e_f}(
  \sigma_{\texttt{pred}=\text{`gunshot\_or\_explosion'}}(M_1) ) \\[2pt]
\text{Gunshot}_C &\equiv \text{Gunshot}_A \cup \text{Gunshot}_B
\end{aligned}
\]
\end{frame}

% ── RQ1: RE-ID ──
\section{RQ1: Re-Identification}
\begin{frame}{RQ1: Object Re-Identification Is Essential}
\begin{columns}[T]
\column{0.48\textwidth}
\small
\textbf{Without re-ID:} VLM assigns new IDs per frame.\\
\textit{Carrying(person=1)} on frame 10,\\
\textit{Carrying(person=7)} on frame 34.\\
Single-frame intervals only.\\
Duration thresholds never satisfied.\\
\alert{Zero detected events.}

\vspace{0.4em}
\textbf{With re-ID:} Two-phase tracking.\\
Frame 1: detect objects and assign IDs.\\
Frame $k$: match by class + block overlap.

\column{0.52\textwidth}
\centering
\footnotesize
\begin{tabular}{lcc}
\toprule
VLM & With re-ID (TP) & Without (TP) \\
\midrule
\textbf{Gemini 3.6 Flash} & \textbf{24} & \textbf{0} \\
Ministral 3-14B & 23 & 0 \\
Gemini 2.5 Flash & 17 & 0 \\
Pixtral 12B & 17 & 0 \\
\bottomrule
\end{tabular}
\vspace{0.3em}

30 scenes. All models: 0 TP without re-ID.
\end{columns}
\end{frame}

% ── CONTRIBUTIONS ──
\section{Contributions}
\begin{frame}{Contributions}
\begin{enumerate}
  \item \textbf{First LALM for surveillance forensics}\\
  Qwen2-Audio-7B: perfect precision (1.000), micro F1 = 0.788 vs.\ PANNs 0.429
  \vspace{0.3em}

  \item \textbf{Object re-identification fixes zero-event bug}\\
  All 4 VLM models: 0 TP without re-ID, up to 0.839 F1 with re-ID
  \vspace{0.3em}

  \item \textbf{First labeled multimodal forensic dataset}\\
  30 scenes, 6 event types, per-frame visual + audio + verdict ground truth
  \vspace{0.3em}

  \item \textbf{Union-based multimodal design}\\
  $C = A \cup B$ preserves all detections; temporal joins reduce recall
  \vspace{0.3em}

  \item \textbf{Fully auditable pipeline}\\
  Every detection = output of a readable SQL query, no black-box reasoning
\end{enumerate}
\end{frame}
